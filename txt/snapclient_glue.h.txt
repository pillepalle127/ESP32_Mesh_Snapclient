#pragma once

#include <stdbool.h>
#include <stdint.h>

#include "esp_err.h"

#ifdef __cplusplus
extern "C" {
#endif

typedef void (*snapclient_recovery_cb_t)(void);

esp_err_t snapclient_start(const char *host, uint16_t port);
void snapclient_pause(bool pause);

/*
 * Registers a non-blocking callback that is invoked after repeated TCP
 * connection failures. The callback must return quickly and must not perform
 * a long-running Wi-Fi recovery directly in the Snapclient task.
 */
void snapclient_set_recovery_callback(snapclient_recovery_cb_t callback);

#ifdef __cplusplus
}
#endif
