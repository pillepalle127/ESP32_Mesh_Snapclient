static const char *TAG = "snap";

typedef enum {
    SNAP_MSG_BASE = 0,
    SNAP_MSG_CODEC_HEADER = 1,
    SNAP_MSG_WIRE_CHUNK = 2,
    SNAP_MSG_SERVER_SETTINGS = 3,
    SNAP_MSG_TIME = 4,
    SNAP_MSG_HELLO = 5,
    SNAP_MSG_STREAM_TAGS = 6,
} snap_msg_type_t;

typedef struct __attribute__((packed)) {
    uint16_t type;
    uint16_t id;
    uint16_t refers_to;
    int32_t sent_sec;
    int32_t sent_usec;
    int32_t received_sec;
    int32_t received_usec;
    uint32_t size;
} snap_base_t;

typedef enum {
    READ_OK = 0,
    READ_TIMEOUT = 1,
    READ_CLOSED = -1,
    READ_ERROR = -2,
} read_result_t;

#define EVT_RESUME                 BIT0
#define SNAP_SAMPLE_RATE           48000
#define SNAP_CHANNELS              2
#define OPUS_MAX_FRAME_SAMPLES     5760
#define OPUS_PCM_SAMPLE_COUNT      (OPUS_MAX_FRAME_SAMPLES * SNAP_CHANNELS)
#define OPUS_PCM_BUFFER_BYTES      (OPUS_PCM_SAMPLE_COUNT * sizeof(opus_int16))
#define SNAP_MESSAGE_BUFFER_SIZE   8192U
#define SNAP_DISCARD_BUFFER_SIZE   2048U
#define SNAP_MAX_MESSAGE_SIZE      (64U * 1024U)
#define SNAP_TASK_STACK_SIZE       12288U
#define SNAP_TASK_PRIORITY         5U
#define SNAP_TASK_CORE             1
#define SNAP_CONNECT_RETRY_MS      2000U
#define SNAP_RECONNECT_DELAY_MS    1000U
#define SNAP_STATS_INTERVAL_US     5000000LL
#define SNAP_TIME_INTERVAL_US      1000000LL
#define SNAP_SOCKET_TIMEOUT_MS     500U
#define SNAP_PARTIAL_TIMEOUT_US    2000000LL
#define SNAP_YIELD_AFTER_MESSAGES  8U

/*
 * Connect timeout.
 *
 * A blocking connect() waits for the lwIP ARP/SYN timeout before it reports
 * EHOSTUNREACH, which took roughly 18-20 s per attempt. That happens whenever
 * the route to the server is temporarily gone, for example while the iot
 * bridge reconfigures its netifs after a station joined the local SoftAP.
 * With a non-blocking connect plus select() a dead route is detected within
 * SNAP_CONNECT_TIMEOUT_MS, so the retry cadence stays predictable.
 */
#define SNAP_CONNECT_TIMEOUT_MS    8000U
#define SNAP_CONNECT_FAILURES_BEFORE_RECOVERY  1U
#define SNAP_RECOVERY_COOLDOWN_US               30000000LL

/*
 * Receive watchdog.
 *
 * When the server restarts, the TCP connection usually does not close
 * cleanly: no FIN arrives, the socket just goes stale. recv() then keeps
 * returning EAGAIN and read_full() reports READ_TIMEOUT forever, so the
 * connection loop would spin indefinitely and never reach the reconnect
 * path in snap_task(). A running server sends wire chunks every 20 ms and
 * answers every time request, so several seconds of complete silence mean
 * the peer is gone.
 */
#define SNAP_RX_WATCHDOG_US        5000000LL

/*
 * TCP keepalive so that lwIP itself can detect a vanished peer even while
 * the application is only waiting for data.
 */
#define SNAP_KEEPALIVE_IDLE_S      3
#define SNAP_KEEPALIVE_INTERVAL_S  2
#define SNAP_KEEPALIVE_COUNT       3

static volatile bool s_run = true;

static bool s_snap_audio_started = false;

static uint16_t s_port = 1704;
static uint16_t s_next_message_id = 1;
static char s_codec[16] = {0};

void snapclient_set_recovery_callback(snapclient_recovery_cb_t callback)
{
    s_recovery_callback = callback;
}

static void maybe_request_network_recovery(void)
{
    if (s_recovery_callback == NULL ||
        s_consecutive_connect_failures <
            SNAP_CONNECT_FAILURES_BEFORE_RECOVERY) {
        return;
    }

    const int64_t now_us = esp_timer_get_time();
    if (s_last_recovery_us != 0 &&
        now_us - s_last_recovery_us < SNAP_RECOVERY_COOLDOWN_US) {
        return;
    }

    s_last_recovery_us = now_us;
    s_consecutive_connect_failures = 0;

    ESP_LOGW(TAG,
             "Repeated Snapserver connection failures; requesting network recovery");
    s_recovery_callback();
}

static int send_full(int fd, const void *buffer, size_t length)
{
    const uint8_t *data = (const uint8_t *)buffer;
    size_t total = 0;
    while (total < length) {
        int rc = send(fd, data + total, length - total, 0);
        if (rc < 0 && errno == EINTR) continue;
        if (rc <= 0) return -1;
        total += (size_t)rc;
    }
    return 0;
}

static read_result_t read_full(int fd, void *buffer, size_t length)
{
    uint8_t *data = (uint8_t *)buffer;
    size_t total = 0;
    int64_t partial_started_us = 0;

    while (total < length && s_run && !s_paused) {
        int rc = recv(fd, data + total, length - total, 0);
        if (rc > 0) {
            if (total == 0) partial_started_us = esp_timer_get_time();
            total += (size_t)rc;
            continue;
        }
        if (rc == 0) return READ_CLOSED;
        if (errno == EINTR) continue;
        if (errno == EAGAIN || errno == EWOULDBLOCK) {
            if (total == 0) return READ_TIMEOUT;
            if (esp_timer_get_time() - partial_started_us >= SNAP_PARTIAL_TIMEOUT_US) {
                return READ_ERROR;
            }
            continue;
        }
        return READ_ERROR;
    }
    return total == length ? READ_OK : READ_ERROR;
}

static void set_socket_timeouts(int fd)
{
    struct timeval timeout = {
        .tv_sec = SNAP_SOCKET_TIMEOUT_MS / 1000,
        .tv_usec = (SNAP_SOCKET_TIMEOUT_MS % 1000) * 1000,
    };
    (void)setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &timeout, sizeof(timeout));
    (void)setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &timeout, sizeof(timeout));

    /*
     * Let the stack probe the peer as well. Without keepalive a server that
     * vanished without sending FIN keeps the socket alive for minutes.
     */
    int keepalive = 1;
    int idle = SNAP_KEEPALIVE_IDLE_S;
    int interval = SNAP_KEEPALIVE_INTERVAL_S;
    int count = SNAP_KEEPALIVE_COUNT;

    (void)setsockopt(fd, SOL_SOCKET, SO_KEEPALIVE, &keepalive, sizeof(keepalive));
    (void)setsockopt(fd, IPPROTO_TCP, TCP_KEEPIDLE, &idle, sizeof(idle));
    (void)setsockopt(fd, IPPROTO_TCP, TCP_KEEPINTVL, &interval, sizeof(interval));
    (void)setsockopt(fd, IPPROTO_TCP, TCP_KEEPCNT, &count, sizeof(count));
}

/* Switches a socket between blocking and non-blocking mode. */
static int set_nonblocking(int fd, bool enable)
{
    int flags = fcntl(fd, F_GETFL, 0);
    if (flags < 0) return -1;

    if (enable) {
        flags |= O_NONBLOCK;
    } else {
        flags &= ~O_NONBLOCK;
    }
    return fcntl(fd, F_SETFL, flags);
}

/*
 * Non-blocking connect bounded by SNAP_CONNECT_TIMEOUT_MS.
 * Returns 0 on success, -1 otherwise. errno describes the failure.
 */
static int connect_with_timeout(int fd,
                                const struct sockaddr *address,
                                socklen_t address_len)
{
    if (set_nonblocking(fd, true) != 0) {
        return -1;
    }

    const int64_t started_us = esp_timer_get_time();

    if (connect(fd, address, address_len) == 0) {
        return set_nonblocking(fd, false);
    }

    if (errno != EINPROGRESS) {
        /* Immediate rejection, for example EHOSTUNREACH with no route. */
        return -1;
    }

    fd_set write_set;
    FD_ZERO(&write_set);
    FD_SET(fd, &write_set);

    struct timeval timeout = {
        .tv_sec = SNAP_CONNECT_TIMEOUT_MS / 1000,
        .tv_usec = (SNAP_CONNECT_TIMEOUT_MS % 1000) * 1000,
    };

    const int ready = select(fd + 1, NULL, &write_set, NULL, &timeout);

    if (ready == 0) {
        ESP_LOGW(TAG,
                 "connect() zu %s:%u nach %u ms abgebrochen (Timeout)",
                 s_host,
                 (unsigned)s_port,
                 (unsigned)SNAP_CONNECT_TIMEOUT_MS);
        errno = ETIMEDOUT;
        return -1;
    }
    if (ready < 0) {
        return -1;
    }

    /* select() only reports writability; ask the socket for the result. */
    int so_error = 0;
    socklen_t error_len = sizeof(so_error);
    if (getsockopt(fd, SOL_SOCKET, SO_ERROR, &so_error, &error_len) != 0) {
        return -1;
    }
    if (so_error != 0) {
        errno = so_error;
        return -1;
    }

    ESP_LOGD(TAG,
             "connect() abgeschlossen nach %lld ms",
             (long long)((esp_timer_get_time() - started_us) / 1000));

    return set_nonblocking(fd, false);
}

static int snap_tcp_connect(void)
{
    struct addrinfo hints = {.ai_family = AF_INET, .ai_socktype = SOCK_STREAM};
    struct addrinfo *result = NULL;
    char port_text[8];
    snprintf(port_text, sizeof(port_text), "%u", (unsigned)s_port);

    int gai = getaddrinfo(s_host, port_text, &hints, &result);
    if (gai != 0 || result == NULL) {
        ESP_LOGW(TAG, "getaddrinfo fehlgeschlagen fuer %s", s_host);
        return -1;
    }

    int fd = socket(result->ai_family, result->ai_socktype, result->ai_protocol);
    if (fd < 0) {
        freeaddrinfo(result);
        return -1;
    }

    if (connect_with_timeout(fd, result->ai_addr, result->ai_addrlen) != 0) {
        ESP_LOGW(TAG, "connect() zu %s:%u fehlgeschlagen (errno=%d)",
                 s_host, (unsigned)s_port, errno);
        close(fd);
        freeaddrinfo(result);
        return -1;
    }

    freeaddrinfo(result);
    set_socket_timeouts(fd);
    ESP_LOGI(TAG, "verbunden mit Snapserver %s:%u", s_host, (unsigned)s_port);
    return fd;
}

static void fill_timestamp(int32_t *sec, int32_t *usec)
{
    int64_t now = esp_timer_get_time();
    *sec = (int32_t)(now / 1000000LL);
    *usec = (int32_t)(now % 1000000LL);
}

static int send_time_request(int fd)
{
    snap_base_t header = {0};
    int32_t latency_payload[2] = {0, 0};
    header.type = SNAP_MSG_TIME;
    header.id = s_next_message_id++;
    header.size = sizeof(latency_payload);

    int32_t sent_sec = 0;
    int32_t sent_usec = 0;

    fill_timestamp(&sent_sec, &sent_usec);

    header.sent_sec = sent_sec;
    header.sent_usec = sent_usec;

    if (send_full(fd, &header, sizeof(header)) != 0) return -1;
    return send_full(fd, latency_payload, sizeof(latency_payload));
}

static int send_hello(int fd)
{
    uint8_t mac[6] = {0};
    if (esp_read_mac(mac, ESP_MAC_WIFI_STA) != ESP_OK) return -1;

    char id[18];
    snprintf(id, sizeof(id), "%02X:%02X:%02X:%02X:%02X:%02X",
             mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
    char name[32];
    snprintf(name, sizeof(name), "ESP32-SnapMesh-%02X%02X", mac[4], mac[5]);

    char json[384];
    int n = snprintf(json, sizeof(json),
        "{\"MAC\":\"%s\",\"HostName\":\"%s\",\"Version\":\"0.27.0\","
        "\"ClientName\":\"%s\",\"OS\":\"esp-idf\",\"Arch\":\"xtensa\","
        "\"Instance\":1,\"SnapStreamProtocolVersion\":2}", id, name, name);
    if (n <= 0 || n >= (int)sizeof(json)) return -1;

    uint32_t payload_length = (uint32_t)n;
    snap_base_t header = {0};
    header.type = SNAP_MSG_HELLO;
    header.id = s_next_message_id++;
    header.size = sizeof(payload_length) + payload_length;

    int32_t sent_sec = 0;
    int32_t sent_usec = 0;
    fill_timestamp(&sent_sec, &sent_usec);
    header.sent_sec = sent_sec;
    header.sent_usec = sent_usec;

    if (send_full(fd, &header, sizeof(header)) != 0 ||
        send_full(fd, &payload_length, sizeof(payload_length)) != 0 ||
        send_full(fd, json, payload_length) != 0) return -1;

    ESP_LOGI(TAG, "Snapcast-Hello: Client=%s, ID=%s", name, id);
    return 0;
}

static esp_err_t opus_decoder_prepare(void)
{
    int decoder_bytes = opus_decoder_get_size(SNAP_CHANNELS);
    ESP_LOGI(TAG,
             "Vor Opus-Init: free=%u largest=%u decoder=%d B pcm=%u B",
             (unsigned)heap_caps_get_free_size(MALLOC_CAP_8BIT),
             (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_8BIT),
             decoder_bytes, (unsigned)OPUS_PCM_BUFFER_BYTES);

    if (decoder_bytes <= 0) return ESP_FAIL;

    if (s_opus_decoder_memory == NULL) {
        s_opus_decoder_memory = heap_caps_malloc((size_t)decoder_bytes,
                                                 MALLOC_CAP_INTERNAL |
                                                 MALLOC_CAP_8BIT);
        if (s_opus_decoder_memory == NULL) {
            ESP_LOGE(TAG, "Kein Speicher fuer Opus-Decoder: need=%d free=%u largest=%u",
                     decoder_bytes,
                     (unsigned)heap_caps_get_free_size(MALLOC_CAP_8BIT),
                     (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_8BIT));
            return ESP_ERR_NO_MEM;
        }
        s_opus_decoder = (OpusDecoder *)s_opus_decoder_memory;
        int rc = opus_decoder_init(s_opus_decoder, SNAP_SAMPLE_RATE, SNAP_CHANNELS);
        if (rc != OPUS_OK) {
            ESP_LOGE(TAG, "opus_decoder_init fehlgeschlagen: %d (%s)",
                     rc, opus_strerror(rc));
            heap_caps_free(s_opus_decoder_memory);
            s_opus_decoder_memory = NULL;
            s_opus_decoder = NULL;
            return ESP_FAIL;
        }
    }

    if (s_opus_pcm == NULL) {
        s_opus_pcm = heap_caps_malloc(OPUS_PCM_BUFFER_BYTES,
                                      MALLOC_CAP_INTERNAL | MALLOC_CAP_8BIT);
        if (s_opus_pcm == NULL) {
            ESP_LOGE(TAG, "Kein Speicher fuer PCM-Puffer: need=%u free=%u largest=%u",
                     (unsigned)OPUS_PCM_BUFFER_BYTES,
                     (unsigned)heap_caps_get_free_size(MALLOC_CAP_8BIT),
                     (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_8BIT));
            return ESP_ERR_NO_MEM;
        }
    }

    int rc = opus_decoder_ctl(s_opus_decoder, OPUS_RESET_STATE);
    if (rc != OPUS_OK) {
        ESP_LOGE(TAG, "OPUS_RESET_STATE fehlgeschlagen: %d (%s)",
                 rc, opus_strerror(rc));
        return ESP_FAIL;
    }
    memset(s_opus_pcm, 0, OPUS_PCM_BUFFER_BYTES);
    ESP_LOGI(TAG, "Opus-Decoder bereit: %d Hz, %d Kanaele, PCM-Puffer=%u B",
             SNAP_SAMPLE_RATE, SNAP_CHANNELS, (unsigned)OPUS_PCM_BUFFER_BYTES);
    return ESP_OK;
}

static void opus_decoder_destroy_all(void)
{
    /* opus_decoder_destroy() must not be used for explicitly allocated state. */
    if (s_opus_decoder_memory != NULL) {
        heap_caps_free(s_opus_decoder_memory);
        s_opus_decoder_memory = NULL;
        s_opus_decoder = NULL;
    }
    if (s_opus_pcm != NULL) {
        heap_caps_free(s_opus_pcm);
        s_opus_pcm = NULL;
    }
}

static void reset_stream_stats(void)
{
    s_wire_chunks = 0; s_wire_bytes = 0; s_decoded_frames = 0;
    s_decoded_bytes = 0; s_arbiter_bytes = 0; s_dropped_bytes = 0;
    s_decode_errors = 0; s_msg_settings = 0; s_msg_time = 0;
    s_msg_tags = 0; s_msg_unknown = 0; s_rx_timeouts = 0;
    s_last_stats_us = esp_timer_get_time();
}

static void log_stream_stats(void)
{
    int64_t now = esp_timer_get_time();
    if (now - s_last_stats_us < SNAP_STATS_INTERVAL_US) return;
    ESP_LOGI(TAG,
        "Stream stats/5s: codec=%s packets=%lu wire=%llu B decoded_frames=%lu "
        "decoded=%llu B arbiter=%llu B dropped=%llu B decode_errors=%lu "
        "settings=%lu time=%lu tags=%lu unknown=%lu rx_timeout=%lu src=%d",
        s_codec[0] ? s_codec : "unknown",
        (unsigned long)s_wire_chunks, (unsigned long long)s_wire_bytes,
        (unsigned long)s_decoded_frames, (unsigned long long)s_decoded_bytes,
        (unsigned long long)s_arbiter_bytes, (unsigned long long)s_dropped_bytes,
        (unsigned long)s_decode_errors, (unsigned long)s_msg_settings,
        (unsigned long)s_msg_time, (unsigned long)s_msg_tags,
        (unsigned long)s_msg_unknown, (unsigned long)s_rx_timeouts,
        (int)arbiter_current());
    reset_stream_stats();
}

static void handle_codec_header(const uint8_t *payload, uint32_t size)
{
    if (payload == NULL || size < sizeof(uint32_t)) return;
    uint32_t codec_length = 0;
    memcpy(&codec_length, payload, sizeof(codec_length));
    if (codec_length == 0 || codec_length >= sizeof(s_codec) ||
        sizeof(uint32_t) + codec_length > size) {
        ESP_LOGE(TAG, "Ungueltiger CodecHeader: size=%lu codec_len=%lu",
                 (unsigned long)size, (unsigned long)codec_length);
        return;
    }
    memset(s_codec, 0, sizeof(s_codec));
    memcpy(s_codec, payload + sizeof(uint32_t), codec_length);
    ESP_LOGI(TAG, "CodecHeader: codec=%s", s_codec);
    if (strcmp(s_codec, "opus") == 0 && opus_decoder_prepare() != ESP_OK) {
        s_decode_errors++;
        ESP_LOGE(TAG, "Opus-Decoder konnte nicht initialisiert werden");
    }
}

static bool activate_snap_source(void)
{
    if (s_snap_audio_started) return true;
    arbiter_set_snapcast_active(true);
    for (unsigned i = 0; i < 20; ++i) {
        if (arbiter_current() == SRC_SNAPCAST) {
            s_snap_audio_started = true;
            return true;
        }
        vTaskDelay(pdMS_TO_TICKS(5));
    }
    ESP_LOGW(TAG, "Snapcast-Quelle wurde nicht rechtzeitig aktiv");
    return false;
}

static void handle_wire_chunk(const uint8_t *payload, uint32_t size)
{
    if (payload == NULL || size < 12) return;
    uint32_t audio_length = 0;
    memcpy(&audio_length, payload + 8, sizeof(audio_length));
    if (audio_length == 0 || audio_length > size - 12) {
        ESP_LOGE(TAG, "WireChunk ungueltig: msg=%lu B audio=%lu B",
                 (unsigned long)size, (unsigned long)audio_length);
        return;
    }

    const uint8_t *audio = payload + 12;
    s_wire_chunks++;
    s_wire_bytes += audio_length;
    if (s_paused) return;

    if (strcmp(s_codec, "pcm") == 0) {
        if (!activate_snap_source()) return;
        size_t accepted = arbiter_feed(SRC_SNAPCAST, audio, audio_length);
        s_decoded_frames++; s_decoded_bytes += audio_length;
        s_arbiter_bytes += accepted; s_dropped_bytes += audio_length - accepted;
        return;
    }

    if (strcmp(s_codec, "opus") == 0) {
        if ((s_opus_decoder == NULL || s_opus_pcm == NULL) &&
            opus_decoder_prepare() != ESP_OK) {
            s_decode_errors++;
            return;
        }
        int samples = opus_decode(s_opus_decoder, audio, (opus_int32)audio_length,
                                  s_opus_pcm, OPUS_MAX_FRAME_SAMPLES, 0);
        if (samples < 0) {
            s_decode_errors++;
            ESP_LOGW(TAG, "opus_decode fehlgeschlagen: %s (%d)",
                     opus_strerror(samples), samples);
            return;
        }
        size_t bytes = (size_t)samples * SNAP_CHANNELS * sizeof(opus_int16);
        if (!activate_snap_source()) return;
        size_t accepted = arbiter_feed(SRC_SNAPCAST, s_opus_pcm, bytes);
        s_decoded_frames++; s_decoded_bytes += bytes;
        s_arbiter_bytes += accepted; s_dropped_bytes += bytes - accepted;
        return;
    }
}

static void process_message(const snap_base_t *header, const uint8_t *payload)
{
    switch (header->type) {
    case SNAP_MSG_CODEC_HEADER: handle_codec_header(payload, header->size); break;
    case SNAP_MSG_WIRE_CHUNK: handle_wire_chunk(payload, header->size); break;
    case SNAP_MSG_SERVER_SETTINGS: s_msg_settings++; break;
    case SNAP_MSG_TIME: s_msg_time++; break;
    case SNAP_MSG_STREAM_TAGS: s_msg_tags++; break;
    default:
        s_msg_unknown++;
        ESP_LOGW(TAG, "Unbekannte Nachricht: type=%u size=%lu",
                 (unsigned)header->type, (unsigned long)header->size);
        break;
    }
}

static void connection_loop(int fd)
{
    static uint8_t discard_buffer[SNAP_DISCARD_BUFFER_SIZE];
    static uint8_t message_buffer[SNAP_MESSAGE_BUFFER_SIZE];
    unsigned messages_without_yield = 0;
    int64_t last_time_request_us = 0;
    int64_t last_rx_us = esp_timer_get_time();
    s_snap_audio_started = false;

    while (s_run && !s_paused) {
        int64_t now = esp_timer_get_time();
        if (last_time_request_us == 0 || now - last_time_request_us >= SNAP_TIME_INTERVAL_US) {
            if (send_time_request(fd) != 0) break;
            last_time_request_us = now;
        }

        snap_base_t header;
        read_result_t rr = read_full(fd, &header, sizeof(header));
        if (rr == READ_TIMEOUT) {
            s_rx_timeouts++;
            log_stream_stats();

            /*
             * A live server sends wire chunks continuously. Prolonged silence
             * means the peer is gone even though the socket still looks open,
             * which happens when the server restarts without a clean close.
             * Leave the loop so snap_task() can reconnect.
             */
            const int64_t silent_us = esp_timer_get_time() - last_rx_us;
            if (silent_us >= SNAP_RX_WATCHDOG_US) {
                ESP_LOGW(TAG,
                         "Keine Serverdaten seit %lld ms, Verbindung gilt als tot",
                         (long long)(silent_us / 1000));
                break;
            }
            continue;
        }
        if (rr != READ_OK) break;

        last_rx_us = esp_timer_get_time();

        if (header.size == 0) continue;
        if (header.size > SNAP_MAX_MESSAGE_SIZE) {
            ESP_LOGE(TAG, "Snapcast-Protokollfehler: Payload=%lu B",
                     (unsigned long)header.size);
            break;
        }

        if (header.size > sizeof(message_buffer)) {
            uint32_t remaining = header.size;
            while (remaining > 0) {
                uint32_t part = remaining > sizeof(discard_buffer)
                              ? sizeof(discard_buffer) : remaining;
                rr = read_full(fd, discard_buffer, part);
                if (rr != READ_OK) { remaining = 1; break; }
                remaining -= part;
            }
            if (remaining != 0) break;
            last_rx_us = esp_timer_get_time();
            continue;
        }

        rr = read_full(fd, message_buffer, header.size);
        if (rr != READ_OK) break;

        last_rx_us = esp_timer_get_time();

        process_message(&header, message_buffer);
        log_stream_stats();
        if (++messages_without_yield >= SNAP_YIELD_AFTER_MESSAGES) {
            messages_without_yield = 0;
            taskYIELD();
        }
    }

    s_snap_audio_started = false;
    arbiter_set_snapcast_active(false);
}

static void snap_task(void *argument)
{
    (void)argument;
    while (s_run) {
        if (s_paused) {
            xEventGroupWaitBits(s_evt, EVT_RESUME, pdTRUE, pdTRUE, portMAX_DELAY);
            if (!s_run) break;
            if (s_paused) continue;
        }
        int fd = snap_tcp_connect();
        if (fd < 0) {
            s_consecutive_connect_failures++;
            ESP_LOGW(TAG,
                     "Snapserver connect failure %lu/%u",
                     (unsigned long)s_consecutive_connect_failures,
                     (unsigned)SNAP_CONNECT_FAILURES_BEFORE_RECOVERY);
            maybe_request_network_recovery();
            vTaskDelay(pdMS_TO_TICKS(SNAP_CONNECT_RETRY_MS));
            continue;
        }

        s_consecutive_connect_failures = 0;
        memset(s_codec, 0, sizeof(s_codec));
        reset_stream_stats();
        if (send_hello(fd) == 0) connection_loop(fd);
        shutdown(fd, SHUT_RDWR);
        close(fd);
        if (!s_paused && s_run) {
            ESP_LOGI(TAG, "Verbindung verloren, Reconnect in %u ms",
                     SNAP_RECONNECT_DELAY_MS);
            vTaskDelay(pdMS_TO_TICKS(SNAP_RECONNECT_DELAY_MS));
        }
    }
    arbiter_set_snapcast_active(false);
    opus_decoder_destroy_all();
    s_task_started = false;
    vTaskDelete(NULL);
}

void snapclient_pause(bool pause)
{
    if (pause) {
        if (s_paused) return;
        s_paused = true;
        ESP_LOGI(TAG, "PAUSE: A2DP aktiv");
        return;
    }
    if (!s_paused) return;
    s_paused = false;
    ESP_LOGI(TAG, "RESUME: Snapclient wird wieder verbunden");
    if (s_evt != NULL) xEventGroupSetBits(s_evt, EVT_RESUME);
}

esp_err_t snapclient_start(const char *host, uint16_t port)
{
    if (s_task_started) return ESP_OK;
    if (host == NULL || host[0] == '\0' || port == 0) return ESP_ERR_INVALID_ARG;
    strlcpy(s_host, host, sizeof(s_host));
    s_port = port;
    if (s_evt == NULL) {
        s_evt = xEventGroupCreate();
        if (s_evt == NULL) return ESP_ERR_NO_MEM;
    }
    s_run = true;
    s_paused = false;
    s_task_started = true;
    memset(s_codec, 0, sizeof(s_codec));
    reset_stream_stats();
    BaseType_t rc = xTaskCreatePinnedToCore(
        snap_task, "snap", SNAP_TASK_STACK_SIZE, NULL,
        SNAP_TASK_PRIORITY, NULL, SNAP_TASK_CORE);
    if (rc != pdPASS) {
        s_task_started = false;
        return ESP_ERR_NO_MEM;
    }
    ESP_LOGI(TAG, "Snapclient gestartet (Server %s:%u, Codec automatisch, CPU %d)",
             s_host, (unsigned)s_port, SNAP_TASK_CORE);
    return ESP_OK;
}
