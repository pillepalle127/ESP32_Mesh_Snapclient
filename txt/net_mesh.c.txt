/**
 * @file  net_mesh.c
 * @brief Autarker ESP-Mesh-Lite Snapclient als reiner Child-Knoten.
 */
#include "net_mesh.h"
#include "snapclient_glue.h"
#include "sdkconfig.h"

#include <stdatomic.h>
#include <string.h>

#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_bridge.h"
#include "esp_event.h"
#include "esp_log.h"
#include "esp_mesh_lite.h"
#include "esp_mesh_lite_core.h"
#include "esp_netif.h"
#include "esp_wifi.h"

static const char *TAG = "net_mesh";

static atomic_bool s_snapclient_started = false;
static TaskHandle_t s_recovery_task;

#define RECOVERY_NOTIFY_BIT BIT0
#define RECOVERY_SETTLE_MS  500U

static void request_network_recovery(void)
{
    if (s_recovery_task == NULL) {
        ESP_LOGW(TAG, "Network recovery requested before task initialization");
        return;
    }

    xTaskNotify(s_recovery_task, RECOVERY_NOTIFY_BIT, eSetBits);
}

static void network_recovery_task(void *arg)
{
    (void)arg;

    for (;;) {
        uint32_t notification = 0;
        xTaskNotifyWait(0, UINT32_MAX, &notification, portMAX_DELAY);

        if ((notification & RECOVERY_NOTIFY_BIT) == 0U) {
            continue;
        }

        ESP_LOGW(TAG,
                 "Repeated Snapserver connection failures; restarting STA link");

        const esp_err_t err = esp_wifi_disconnect();
        if (err != ESP_OK && err != ESP_ERR_WIFI_NOT_CONNECT) {
            ESP_LOGW(TAG,
                     "esp_wifi_disconnect failed: %s",
                     esp_err_to_name(err));
        }

        /*
         * Mesh-Lite owns the STA association and starts its normal reconnect
         * sequence after the disconnect event. A short delay prevents a
         * pending TCP retry from racing the Wi-Fi state transition.
         */
        vTaskDelay(pdMS_TO_TICKS(RECOVERY_SETTLE_MS));
    }
}

static void ip_event_handler(void *arg, esp_event_base_t base,
                             int32_t id, void *data)
{
    (void)arg;
    (void)base;

    if (id != IP_EVENT_STA_GOT_IP) {
        return;
    }

    ip_event_got_ip_t *event = (ip_event_got_ip_t *)data;
    ESP_LOGI(TAG, "GOT IP: " IPSTR " (Parent link ready)",
             IP2STR(&event->ip_info.ip));

    /*
     * This node has a STA and a SoftAP netif. Explicitly restore the STA as
     * the default route after every DHCP/reassociation event so traffic for
     * the parent and Snapserver cannot remain assigned to the local SoftAP.
     */
    if (event->esp_netif != NULL) {
        const esp_err_t route_err =
            esp_netif_set_default_netif(event->esp_netif);
        if (route_err != ESP_OK) {
            ESP_LOGW(TAG,
                     "Could not set STA as default netif: %s",
                     esp_err_to_name(route_err));
        } else {
            ESP_LOGI(TAG, "STA netif restored as default route");
        }
    }

    esp_err_t err = esp_wifi_set_ps(WIFI_PS_NONE);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "Could not disable Wi-Fi power save: %s",
                 esp_err_to_name(err));
    } else {
        ESP_LOGI(TAG, "Wi-Fi power save disabled");
    }

    bool expected = false;
    if (!atomic_compare_exchange_strong(&s_snapclient_started,
                                         &expected,
                                         true)) {
        return;
    }

    err = snapclient_start(CONFIG_SNAPSERVER_HOST,
                           CONFIG_SNAPSERVER_PORT);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Snapclient start failed: %s", esp_err_to_name(err));
        atomic_store(&s_snapclient_started, false);
    } else {
        ESP_LOGI(TAG, "Snapclient started after obtaining parent IP");
    }
}

static void mesh_info_task(void *arg)
{
    (void)arg;
    for (;;) {
        const uint8_t level = esp_mesh_lite_get_level();
        ESP_LOGI(TAG, "mesh level=%u  %s",
                 level,
                 (level == 1U) ? "[UNEXPECTED ROOT]"
                               : "[Child - via Parent]");
        vTaskDelay(pdMS_TO_TICKS(10000));
    }
}

static void configure_routerless_wifi(void)
{
    wifi_config_t sta_config = {0};
    esp_bridge_wifi_set_config(WIFI_IF_STA, &sta_config);

    wifi_config_t ap_config = {
        .ap = {
            .password = CONFIG_MESH_SOFTAP_PASSWORD,
            .channel = CONFIG_MESH_CHANNEL,
            .max_connection = 10,
        },
    };

    strlcpy((char *)ap_config.ap.ssid,
            CONFIG_MESH_SOFTAP_SSID_PREFIX,
            sizeof(ap_config.ap.ssid));
    ap_config.ap.ssid_len = strlen((char *)ap_config.ap.ssid);
    ap_config.ap.authmode = strlen(CONFIG_MESH_SOFTAP_PASSWORD) >= 8U
                                ? WIFI_AUTH_WPA2_PSK
                                : WIFI_AUTH_OPEN;

    esp_bridge_wifi_set_config(WIFI_IF_AP, &ap_config);
}

static esp_err_t configure_mesh_softap_info(void)
{
    const esp_err_t err = esp_mesh_lite_set_softap_info(
        CONFIG_MESH_SOFTAP_SSID_PREFIX,
        CONFIG_MESH_SOFTAP_PASSWORD);

    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Could not set Mesh-Lite SoftAP info: %s",
                 esp_err_to_name(err));
        return err;
    }

    ESP_LOGI(TAG,
             "Mesh SoftAP configured: SSID=%s, channel=%d, password=[HIDDEN]",
             CONFIG_MESH_SOFTAP_SSID_PREFIX,
             CONFIG_MESH_CHANNEL);
    return ESP_OK;
}

esp_err_t net_mesh_start(void)
{
    esp_bridge_create_all_netif();
    configure_routerless_wifi();

    if (xTaskCreate(network_recovery_task,
                    "net_recovery",
                    3072,
                    NULL,
                    4,
                    &s_recovery_task) != pdPASS) {
        ESP_LOGE(TAG, "Could not create network recovery task");
        return ESP_ERR_NO_MEM;
    }

    snapclient_set_recovery_callback(request_network_recovery);

    ESP_ERROR_CHECK(esp_event_handler_instance_register(
        IP_EVENT, ESP_EVENT_ANY_ID, &ip_event_handler, NULL, NULL));

    esp_mesh_lite_config_t mesh_config = ESP_MESH_LITE_DEFAULT_INIT();
    mesh_config.join_mesh_ignore_router_status = true;
    mesh_config.join_mesh_without_configured_wifi = true;
    mesh_config.leaf_node = false;

    esp_mesh_lite_init(&mesh_config);

    esp_err_t err = configure_mesh_softap_info();
    if (err != ESP_OK) {
        return err;
    }

    err = esp_mesh_lite_set_disallowed_level(1);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Could not disallow root level 1: %s",
                 esp_err_to_name(err));
        return err;
    }

    esp_mesh_lite_start();

    if (xTaskCreate(mesh_info_task, "mesh_info", 3072, NULL, 3, NULL) != pdPASS) {
        ESP_LOGE(TAG, "Could not create mesh info task");
        return ESP_ERR_NO_MEM;
    }

    ESP_LOGI(TAG,
             "Mesh-Lite started as routerless Child; server=%s:%d",
             CONFIG_SNAPSERVER_HOST,
             CONFIG_SNAPSERVER_PORT);
    return ESP_OK;
}
